// correlationMiddleware propagates and assigns X-Correlation-ID (GEN-08).
func (s *Server) correlationMiddleware(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                cid := r.Header.Get("X-Correlation-ID")
                if cid == "" {
                        cid = uuid.NewString()
                }
                w.Header().Set("X-Correlation-ID", cid)
                ctx := context.WithValue(r.Context(), correlationKey, cid)

                start := time.Now()
                rw := &statusResponseWriter{ResponseWriter: w, status: http.StatusOK}

                next.ServeHTTP(rw, r.WithContext(ctx))

                logJSON("INFO", s.cfg.ServiceName, cid, "HTTP Request completed", map[string]any{
                        "method":      r.Method,
                        "path":        r.URL.Path,
                        "status":      rw.status,
                        "duration_ms": time.Since(start).Milliseconds(),
                })
        })
}

// corsMiddleware sets CORS headers allowing frontend interaction without errors.
func corsMiddleware(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                w.Header().Set("Access-Control-Allow-Origin", "*")
                w.Header().Set("Access-Control-Allow-Methods", "GET, POST, PUT, PATCH, DELETE, OPTIONS")
                w.Header().Set("Access-Control-Allow-Headers", "Content-Type, Authorization, X-Correlation-ID, Idempotency-Key")

                if r.Method == http.MethodOptions {
                        w.WriteHeader(http.StatusNoContent)
                        return
                }
                next.ServeHTTP(w, r)
        })
}

                 OR EXISTS(SELECT 1 FROM exceptions WHERE account_id = $1)`,
                accountUUID).Scan(&accountExists)
        if err != nil {
                writeProblem(w, r, http.StatusInternalServerError, "INTERNAL_ERROR", "Database Error", err.Error())
                return
        }

        if !accountExists {
                writeProblem(w, r, http.StatusNotFound, "NOT_FOUND", "Account Not Found",
                        fmt.Sprintf("Account %s was not found in bank records", accountUUID.String()))
                return
        }

        rows, err := s.db.QueryContext(r.Context(), `
                SELECT id, reference_id, discrepancy_type, expected_amount, actual_amount, difference_amount, currency, status, created_at
                FROM exceptions
                WHERE account_id = $1
                ORDER BY created_at DESC
        `, accountUUID)
        if err != nil {
                writeProblem(w, r, http.StatusInternalServerError, "INTERNAL_ERROR", "Database Error", err.Error())
                return
        }
        defer rows.Close()

        items := make([]AccountExceptionItem, 0)
        for rows.Next() {
                var item AccountExceptionItem
                var diffAmt int64
                if err := rows.Scan(
                        &item.ID,
                        &item.ReferenceID,
                        &item.DiscrepancyType,
                        &item.ExpectedAmount,
                        &item.ActualAmount,
                        &diffAmt,
                        &item.Currency,
                        &item.Status,
                        &item.CreatedAt,
                ); err != nil {
                        writeProblem(w, r, http.StatusInternalServerError, "INTERNAL_ERROR", "Scan Error", err.Error())
                        return
                }
                item.Difference = diffAmt
                items = append(items, item)
        }

        w.Header().Set("Content-Type", "application/json")
        _ = json.NewEncoder(w).Encode(items)
}

// Pagination limits for GET /api/v1/exceptions (GEN-09).
const (
        defaultExceptionsLimit = 50
        maxExceptionsLimit     = 200
)

// Allowed values of the ?status= filter.
var validExceptionStatuses = map[string]bool{
        "OPEN":         true,
        "UNDER_REVIEW": true,
        "RESOLVED":     true,
        "DISMISSED":    true,
}

// exceptionCursor is the position after the last item of a page (keyset pagination).
type exceptionCursor struct {
        CreatedAt time.Time `json:"created_at"`
        ID        string    `json:"id"`
}

// encode returns an opaque URL-safe token: base64url of {"created_at": ..., "id": ...}.
func (c exceptionCursor) encode() string {
        raw, _ := json.Marshal(c)
        return base64.RawURLEncoding.EncodeToString(raw)
}

func decodeExceptionCursor(token string) (exceptionCursor, error) {
        var c exceptionCursor
        raw, err := base64.RawURLEncoding.DecodeString(token)
        if err != nil {
                return c, fmt.Errorf("not valid base64url")
        }
        if err := json.Unmarshal(raw, &c); err != nil {
                return c, fmt.Errorf("not valid JSON")
        }
        if c.CreatedAt.IsZero() {
                return c, fmt.Errorf("created_at is missing")
        }
        if _, err := uuid.Parse(c.ID); err != nil {
                return c, fmt.Errorf("id is not a UUID")
        }
        return c, nil
}

// ListExceptionsHandler godoc
// @Summary      List discrepancies and exceptions
// @Description  Paginated list of financial discrepancies, oldest first (created_at, id). Cursor pagination (GEN-09).
// @Tags         Reconciliation Exceptions
// @Produce      json
// @Param        status  query  string  false  "OPEN | UNDER_REVIEW | RESOLVED | DISMISSED"
// @Param        limit   query  int     false  "1..200, default 50; above 200 returns 400"
// @Param        cursor  query  string  false  "next_cursor of the previous page"
// @Success      200  {object}  ExceptionsResponse
// @Failure      400  {object}  ProblemDetails
// @Failure      500  {object}  ProblemDetails
// @Router       /api/v1/exceptions [get]
func (s *Server) ListExceptionsHandler(w http.ResponseWriter, r *http.Request) {
        q := r.URL.Query()

        // ---- 1. validate the query string ----
        limit := defaultExceptionsLimit
        if v := q.Get("limit"); v != "" {
                n, err := strconv.Atoi(v)
                if err != nil || n < 1 {
                        writeProblem(w, r, http.StatusBadRequest, "VALIDATION_ERROR", "Invalid limit",
                                "limit must be a positive integer")
                        return
                }
                if n > maxExceptionsLimit {
                        writeProblem(w, r, http.StatusBadRequest, "VALIDATION_ERROR", "Invalid limit",
                                fmt.Sprintf("limit must not exceed %d", maxExceptionsLimit))
                        return
                }
                limit = n
        }

        status := q.Get("status")
        if status != "" && !validExceptionStatuses[status] {
                writeProblem(w, r, http.StatusBadRequest, "VALIDATION_ERROR", "Invalid status",
                        "status must be one of OPEN, UNDER_REVIEW, RESOLVED, DISMISSED")
                return
        }

        var cursor *exceptionCursor
        if v := q.Get("cursor"); v != "" {
                c, err := decodeExceptionCursor(v)
                if err != nil {
                        writeProblem(w, r, http.StatusBadRequest, "VALIDATION_ERROR", "Invalid cursor",
                                "cursor is not a valid token: "+err.Error())
                        return
                }
                cursor = &c
        }

        // ---- 2. build the query with dynamic WHERE clauses ----
        var (
                conds []string
                args  []any
        )
        if status != "" {
                args = append(args, status)
                conds = append(conds, fmt.Sprintf("status = $%d", len(args)))
        }
        if cursor != nil {
                // keyset pagination: everything after the last row of the previous page
                args = append(args, cursor.CreatedAt, cursor.ID)
                conds = append(conds, fmt.Sprintf("(created_at, id) > ($%d::timestamptz, $%d::uuid)", len(args)-1, len(args)))
        }

        query := `
                SELECT id,
                       COALESCE(job_id::text, ''),
                       reference_id::text,
                       CASE discrepancy_type
                            WHEN 'MISSING_IN_SETTLEMENT' THEN 'MISSING_IN_GATEWAY'
                            WHEN 'STATUS_CONFLICT'       THEN 'STATUS_MISMATCH'
                            ELSE discrepancy_type
                       END,
                       status,
                       expected_amount, actual_amount, difference_amount,
                       format('%s: expected %s %s, actual %s', discrepancy_type, expected_amount, currency, actual_amount),
                       resolution_notes, NULL::text, created_at
                FROM exceptions`
        if len(conds) > 0 {
                query += " WHERE " + strings.Join(conds, " AND ")
        }
        args = append(args, limit+1) // one extra row tells us whether a next page exists
        query += fmt.Sprintf(" ORDER BY created_at ASC, id ASC LIMIT $%d", len(args))

        rows, err := s.db.QueryContext(r.Context(), query, args...)
        if err != nil {
                logJSON("ERROR", s.cfg.ServiceName, correlationID(r), "list exceptions query failed", map[string]any{"error": err.Error()})
                writeProblem(w, r, http.StatusInternalServerError, "INTERNAL_ERROR", "Database Error", "failed to query exceptions")
                return
        }
        defer rows.Close()

        items := make([]Discrepancy, 0, limit)
        for rows.Next() {
                var d Discrepancy
                var ledgerAmt, gatewayAmt, diffAmt int64
                if err := rows.Scan(&d.ID, &d.JobID, &d.TransactionReference, &d.Type, &d.Status,
                        &ledgerAmt, &gatewayAmt, &diffAmt, &d.Description,
                        &d.ResolutionNotes, &d.ResolvedBy, &d.CreatedAt); err != nil {
                        logJSON("ERROR", s.cfg.ServiceName, correlationID(r), "list exceptions scan failed", map[string]any{"error": err.Error()})
                        writeProblem(w, r, http.StatusInternalServerError, "INTERNAL_ERROR", "Database Error", "failed to read exceptions")
                        return
                }
                d.LedgerAmount = Money{Amount: ledgerAmt, Currency: "IRR"}
                d.GatewayAmount = Money{Amount: gatewayAmt, Currency: "IRR"}
                d.DifferenceAmount = Money{Amount: diffAmt, Currency: "IRR"}
                items = append(items, d)
        }
        if err := rows.Err(); err != nil {
                logJSON("ERROR", s.cfg.ServiceName, correlationID(r), "list exceptions iteration failed", map[string]any{"error": err.Error()})
                writeProblem(w, r, http.StatusInternalServerError, "INTERNAL_ERROR", "Database Error", "failed to read exceptions")
                return
        }

        // ---- 3. build next_cursor ----
        var next *string
        if len(items) > limit {
                items = items[:limit]
                last := items[limit-1]
                token := exceptionCursor{CreatedAt: last.CreatedAt, ID: last.ID}.encode()
                next = &token
        }

        w.Header().Set("Content-Type", "application/json")
        _ = json.NewEncoder(w).Encode(ExceptionsResponse{Items: items, NextCursor: next})
}

// DefaultS2BaseURL is the default location of the S2 Ledger service.
const DefaultS2BaseURL = "http://localhost:8082"

// S2ClientConfig holds configuration for the S2 Ledger client.
type S2ClientConfig struct {
        BaseURL string
        Timeout time.Duration
}

// LoadS2Config loads S2 client settings from environment variables.
func LoadS2Config() S2ClientConfig {
        baseURL := os.Getenv("NEIGHBOR_BASE_URL")
        if baseURL == "" {
                baseURL = os.Getenv("S2_LEDGER_BASE_URL")
        }
        if baseURL == "" {
                baseURL = DefaultS2BaseURL
        }

        return S2ClientConfig{
                BaseURL: strings.TrimRight(baseURL, "/"),
                Timeout: 5 * time.Second,
        }
}

// S2Client communicates with Service 2 (Ledger & Balance - S2-LDG).
type S2Client struct {
        baseURL    string
        httpClient *http.Client
}

// NewS2Client initializes a new S2Client with connection pooling and timeouts.
func NewS2Client(cfg S2ClientConfig) *S2Client {
        timeout := cfg.Timeout
        if timeout <= 0 {
                timeout = 5 * time.Second
        }

        transport := &http.Transport{
                Proxy: http.ProxyFromEnvironment,
                DialContext: (&net.Dialer{
                        Timeout:   3 * time.Second,
                        KeepAlive: 30 * time.Second,
                }).DialContext,
                MaxIdleConns:        100,
                MaxIdleConnsPerHost: 20,
                IdleConnTimeout:     90 * time.Second,
        }

        return &S2Client{
                baseURL: strings.TrimRight(cfg.BaseURL, "/"),
                httpClient: &http.Client{
                        Transport: transport,
                        Timeout:   timeout,
                },
        }
}

// --- Fault Tolerance & Error Handling ---

// S2ProblemError represents RFC 7807/9457 error details returned by S2.
type S2ProblemError struct {
        StatusCode int
        Code       string
        Title      string
        Detail     string
        Problem    ProblemDetails
}

func (e *S2ProblemError) Error() string {
        return fmt.Sprintf("s2 error (status %d, code %s): %s - %s", e.StatusCode, e.Code, e.Title, e.Detail)
}

// Client error categories
var (
        ErrUpstreamTimeout    = errors.New("UPSTREAM_TIMEOUT: upstream S2 service request timed out")
        ErrServiceUnavailable = errors.New("SERVICE_UNAVAILABLE: upstream S2 service is unreachable")
        ErrTrialBalanceFailed = errors.New("TRIAL_BALANCE_UNBALANCED: S2 ledger total debits do not equal total credits")
)

func (c *S2Client) handleRequestError(err error) error {
        if err == nil {
                return nil
        }
        var netErr net.Error
        if errors.As(err, &netErr) && netErr.Timeout() {
                return fmt.Errorf("%w: %v", ErrUpstreamTimeout, err)
        }
        if errors.Is(err, context.DeadlineExceeded) {
                return fmt.Errorf("%w: %v", ErrUpstreamTimeout, err)
        }
        return fmt.Errorf("%w: %v", ErrServiceUnavailable, err)
}

func (c *S2Client) doRequest(req *http.Request) (*http.Response, error) {
        // Propagate X-Correlation-ID from context (GEN-08)
        if cid, ok := req.Context().Value(correlationKey).(string); ok && cid != "" {
                req.Header.Set("X-Correlation-ID", cid)
        }
        req.Header.Set("Accept", "application/json, application/problem+json")

        resp, err := c.httpClient.Do(req)
        if err != nil {
                return nil, c.handleRequestError(err)
        }
        return resp, nil
}

// --- Client Methods ---

// FetchJournalEntries queries S2 for ledger entries matching the query criteria.
// Target: GET /api/v1/journal-entries
func (c *S2Client) FetchJournalEntries(ctx context.Context, query JournalEntryQuery) (*PagedJournalEntries, error) {
        endpoint := fmt.Sprintf("%s/api/v1/journal-entries", c.baseURL)

        params := url.Values{}
        if query.From != nil {
                params.Set("from", query.From.UTC().Format(time.RFC3339))
        }
        if query.To != nil {
                params.Set("to", query.To.UTC().Format(time.RFC3339))
        }
        limit := query.Limit
        if limit <= 0 {
                limit = 50
        } else if limit > 200 {
                limit = 200
        }
        params.Set("limit", strconv.Itoa(limit))

        if query.Cursor != "" {
                params.Set("cursor", query.Cursor)
        }
        if query.AccountID != "" {
                params.Set("account_id", query.AccountID)
        }
        if query.ReferenceID != "" {
                params.Set("reference_id", query.ReferenceID)
        }

        if len(params) > 0 {
                endpoint = fmt.Sprintf("%s?%s", endpoint, params.Encode())
        }

        req, err := http.NewRequestWithContext(ctx, http.MethodGet, endpoint, nil)
        if err != nil {
                return nil, fmt.Errorf("create request: %w", err)
        }

        resp, err := c.doRequest(req)
        if err != nil {
                return nil, err
        }
        defer resp.Body.Close()

        bodyBytes, err := io.ReadAll(resp.Body)
        if err != nil {
                return nil, fmt.Errorf("read response body: %w", err)
        }

        if resp.StatusCode >= 400 {
                var prob ProblemDetails
                if err := json.Unmarshal(bodyBytes, &prob); err == nil && prob.Title != "" {
                        return nil, &S2ProblemError{
                                StatusCode: resp.StatusCode,
                                Code:       prob.Code,
                                Title:      prob.Title,
                                Detail:     prob.Detail,
                                Problem:    prob,
                        }
                }
                return nil, &S2ProblemError{
                        StatusCode: resp.StatusCode,
                        Code:       "UPSTREAM_ERROR",
                        Title:      "Upstream S2 Error",
                        Detail:     string(bodyBytes),
                }
        }

        var result PagedJournalEntries
        if err := json.Unmarshal(bodyBytes, &result); err != nil {
                return nil, fmt.Errorf("unmarshal paged journal entries: %w", err)
        }

        return &result, nil
}

// FetchAllJournalEntries collects all journal entries in a date range by following cursor pagination.
func (c *S2Client) FetchAllJournalEntries(ctx context.Context, query JournalEntryQuery) ([]JournalEntryDto, error) {
        var allEntries []JournalEntryDto
        currentQuery := query

        for {
                page, err := c.FetchJournalEntries(ctx, currentQuery)
                if err != nil {
                        return nil, err
                }

                allEntries = append(allEntries, page.Items...)

                if page.NextCursor == nil || *page.NextCursor == "" {
                        break
                }
                currentQuery.Cursor = *page.NextCursor
        }

        return allEntries, nil
}

// FetchTrialBalance queries S2 to verify ledger integrity (sum debits == sum credits).
// Target: GET /api/v1/reports/trial-balance
func (c *S2Client) FetchTrialBalance(ctx context.Context) (*TrialBalanceResponse, error) {
        endpoint := fmt.Sprintf("%s/api/v1/reports/trial-balance", c.baseURL)

        req, err := http.NewRequestWithContext(ctx, http.MethodGet, endpoint, nil)
        if err != nil {
                return nil, fmt.Errorf("create request: %w", err)
        }

        resp, err := c.doRequest(req)
        if err != nil {
                return nil, err
        }
        defer resp.Body.Close()

        bodyBytes, err := io.ReadAll(resp.Body)
        if err != nil {
                return nil, fmt.Errorf("read response body: %w", err)
        }

        if resp.StatusCode >= 400 {
                var prob ProblemDetails
                if err := json.Unmarshal(bodyBytes, &prob); err == nil && prob.Title != "" {
                        return nil, &S2ProblemError{
                                StatusCode: resp.StatusCode,
                                Code:       prob.Code,
                                Title:      prob.Title,
                                Detail:     prob.Detail,
                                Problem:    prob,
                        }
                }
                return nil, &S2ProblemError{
                        StatusCode: resp.StatusCode,
                        Code:       "UPSTREAM_ERROR",
                        Title:      "Upstream S2 Error",
                        Detail:     string(bodyBytes),
                }
        }

        var tb TrialBalanceResponse
        if err := json.Unmarshal(bodyBytes, &tb); err != nil {
                return nil, fmt.Errorf("unmarshal trial balance: %w", err)
        }

        // Verify integrity checks
        if !tb.IsBalanced || tb.TotalDebit != tb.TotalCredit {
                return &tb, fmt.Errorf("%w: total_debit=%d total_credit=%d is_balanced=%t",
                        ErrTrialBalanceFailed, tb.TotalDebit, tb.TotalCredit, tb.IsBalanced)
        }

        return &tb, nil
}
